   49: =head2 new
   50: 
   51: Purpose: start a server for one session.
   52: 
   53: Args: C<root> (the server root), C<repo>, C<client>, optional C<role>
   54: (default C<client>) and C<server_bin> (default C<FILESYNC_SERVER_BIN>, then a
   55: F<bin/filesync-server> found above this module, then C<filesync-server> on
   56: the PATH).
   57: 
   58: Returns: the transport.
   59: 
   60: Side Effects: starts C<$^X server_bin --root ... --repo ... --client ...
   61: --role ...>.  The child's standard error is the parent's, never the protocol
   62: pipe: sharing it would put diagnostics into the protocol stream.
   63: 
   64: Usage:
   65: 
   66: 	my $transport = App::FileSync::Transport::Local->new(%args);
   67: 
   68: =head3 EXAMPLE
   69: 
   70: 	my $t = App::FileSync::Transport::Local->new(root => $root, repo => 'docs', client => 'test-1');
   71: 	print {$t->writer} ...;
   72:

  100: sub new {
  101: 	my $class = shift;
  102: 	my $args = named(\@_, {
  103: 		map({ $_ => { type => 'string', optional => 1 } } qw(root repo client role server_bin)),
  104: 	}, 'Transport::Local->new');
  105: 
  106: 	for my $required (qw(root repo client)) {
  107: 		throw('transport.missing', { transport => 'Local', what => $required })
  108: 			unless defined $args->{$required} && length $args->{$required};
  109: 	}
  110: 
  111: 	my @command = (
  112: 		$^X, $args->{server_bin} || _default_server_bin(),
  113: 		'--root',   $args->{root},
  114: 		'--repo',   $args->{repo},
  115: 		'--client', $args->{client},
  116: 		'--role',   $args->{role} || $ROLE_CLIENT,
  117: 	);
  118: 
  119: 	# The child's stderr is a dup of ours.  A pipe here would not be read, and
  120: 	# the first log line would kill the server with SIGPIPE.
  121: 	my $child = App::FileSync::Process::spawn(command => \@command, stderr => fileno(STDERR));
  122: 
  123: 	return bless { %{$child} }, $class;
  124: }

  180: =head2 close
  181: 
  182: Purpose: end the session's process.
  183: 
  184: Args: none.
  185: 
  186: Returns: 1.  Repeated calls do nothing.
  187: 
  188: Side Effects: closes both pipes (the server sees end of file) and reaps the
  189: child.  A server that has not exited after
  190: C<$App::FileSync::Process::GRACE_SECONDS> is stopped, so a server that never
  191: sees end of file cannot hang the client.
  192: 
  193: Usage:
  194: 
  195: 	$transport->close;
  196: 
  197: =head3 EXAMPLE
  198: 
  199: 	$transport->close;	# safe to call twice

  221: sub close {	## no critic (Subroutines::ProhibitBuiltinHomonyms, Subroutines::ProhibitAmbiguousNames) -- transport interface
  222: 	my ($self) = @_;
  223: 
  224: 	return 1 if $self->{closed}++;
  225: 
  226: 	# Pipe close failures mean the child already went; reaping still matters.
  227: 	for my $handle (grep { defined } @{$self}{qw(writer reader)}) {
  228: 		CORE::close($handle);
  229: 	}
  230: 	if ($self->{pid}) {
  231: 		$self->{end} = App::FileSync::Process::reap(pid => $self->{pid});
  232: 		$self->{pid} = 0;
  233: 	}
  234: 
  235: 	return 1;
  236: }

  281: # _default_server_bin
  282: # Purpose:     find the server program when none is configured.
  283: # Entry:       none.
  284: # Exit:        returns FILESYNC_SERVER_BIN, a bin/filesync-server found by
  285: #              walking up from this file (a checkout, or blib under
  286: #              "make test"), or the bare name (installed, on the PATH).
  287: # Side effects: none.
  288: sub _default_server_bin :Private {
  289: 	my $configured = $ENV{FILESYNC_SERVER_BIN};
  290: 	return $configured if defined $configured && length $configured;
  291: 
  292: 	my $directory = dirname(__FILE__);
  293: 	for (1 .. $SEARCH_DEPTH) {
  294: 		my $candidate = File::Spec->catfile($directory, 'bin', $SERVER_PROGRAM);
  295: 		return $candidate if -f $candidate;
  296: 		$directory = dirname($directory);
  297: 	}
  298: 
  299: 	return $SERVER_PROGRAM;
  300: }
