   49: =head2 new
   50: 
   51: Purpose: start a server for one session.
   52: 
   53: Args: C<root> (the server root), C<repo>, C<client>, optional C<role>
   54: (default C<client>) and C<server_bin> (default C<FILESYNC_SERVER_BIN>, then a
   55: F<bin/filesync-server> found above this module, then C<filesync-server> on
   56: the PATH).
   57: 
   58: Returns: the transport.
   59: 
   60: Side Effects: starts C<$^X server_bin --root ... --repo ... --client ...
   61: --role ...>.  The child's standard error is the parent's, never the protocol
   62: pipe: sharing it would put diagnostics into the protocol stream.
   63: 
   64: Usage:
   65: 
   66: 	my $transport = App::FileSync::Transport::Local->new(%args);
   67: 
   68: =head3 EXAMPLE
   69: 
   70: 	my $t = App::FileSync::Transport::Local->new(root => $root, repo => 'docs', client => 'test-1');
   71: 	print {$t->writer} ...;
   72:

  100: sub new {
  101: 	my $class = shift;
  102: 	my $args = named(\@_, {
  103: 		map({ $_ => { type => 'string', optional => 1 } } qw(root repo client role server_bin)),
  104: 	}, 'Transport::Local->new');
  105: 
  106: 	for my $required (qw(root repo client)) {
  107: 		throw('transport.missing', { transport => 'Local', what => $required })
  108: 			unless defined $args->{$required} && length $args->{$required};
  109: 	}
  110: 
  111: 	my @command = (
  112: 		$^X, $args->{server_bin} || _default_server_bin(),
  113: 		'--root',   $args->{root},
  114: 		'--repo',   $args->{repo},
  115: 		'--client', $args->{client},
  116: 		'--role',   $args->{role} || $ROLE_CLIENT,
  117: 	);
  118: 
  119: 	# '>&STDERR' dups the parent's stderr.  Passing a filehandle here would not
  120: 	# redirect the child at all: it would write into an unread pipe, and the
  121: 	# first log line would kill it with SIGPIPE.
  122: 	my ($writer, $reader);
  123: 	my $pid = IPC::Open3::open3($writer, $reader, '>&STDERR', @command);
  124: 
  125: 	return bless { pid => $pid, writer => $writer, reader => $reader }, $class;
  126: }

  221: sub close {	## no critic (Subroutines::ProhibitBuiltinHomonyms, Subroutines::ProhibitAmbiguousNames) -- transport interface
  222: 	my ($self) = @_;
  223: 
  224: 	return 1 if $self->{closed}++;
  225: 
  226: 	# Pipe close failures mean the child already went; reaping still matters.
  227: 	for my $handle (grep { defined } @{$self}{qw(writer reader)}) {
  228: 		CORE::close($handle);
  229: 	}
  230: 	waitpid($self->{pid}, 0) if $self->{pid};
  231: 
  232: 	return 1;
  233: }

  278: # _default_server_bin
  279: # Purpose:     find the server program when none is configured.
  280: # Entry:       none.
  281: # Exit:        returns FILESYNC_SERVER_BIN, a bin/filesync-server found by
  282: #              walking up from this file (a checkout, or blib under
  283: #              "make test"), or the bare name (installed, on the PATH).
  284: # Side effects: none.
  285: sub _default_server_bin :Private {
  286: 	my $configured = $ENV{FILESYNC_SERVER_BIN};
  287: 	return $configured if defined $configured && length $configured;
  288: 
  289: 	my $directory = dirname(__FILE__);
  290: 	for (1 .. $SEARCH_DEPTH) {
  291: 		my $candidate = File::Spec->catfile($directory, 'bin', $SERVER_PROGRAM);
  292: 		return $candidate if -f $candidate;
  293: 		$directory = dirname($directory);
  294: 	}
  295: 
  296: 	return $SERVER_PROGRAM;
  297: }
