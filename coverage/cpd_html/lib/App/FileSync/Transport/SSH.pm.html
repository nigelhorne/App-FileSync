   79: =head2 new
   80: 
   81: Purpose: start C<filesync-server> on the server host for one session.
   82: 
   83: Args: C<server> (C<user@host>, C<host> or C<user@[v6]>), C<repo>, C<client>,
   84: optional C<role> (default C<client>), C<server_command> (default
   85: C<filesync-server>), C<host_key_checking> (C<yes>, the default, or
   86: C<accept-new>), C<known_hosts>, C<identity>, C<port>, C<command> (the local
   87: ssh program), C<extra_args> (arrayref of extra ssh arguments).
   88: 
   89: Returns: the transport.
   90: 
   91: Side Effects: runs C<ssh ... SERVER PROGRAM --repo R --client C --role X
   92: --proto 2>.  ssh's standard error goes to a temporary file, never the
   93: protocol pipe, and is available from L</error_text>.  On Windows a C<.cmd> or
   94: C<.bat> ssh wrapper is started through C<COMSPEC>, since CreateProcess cannot
   95: start one directly.
   96: 
   97: Usage:
   98: 
   99: 	my $transport = App::FileSync::Transport::SSH->new(%args);
  100: 
  101: =head3 EXAMPLE
  102: 
  103: 	my $t = App::FileSync::Transport::SSH->new(
  104: 		server => 'backup', repo => 'photos', client => 'nas', role => 'replica', port => 2222,
  105: 	);
  106: 	print join(' ', @{ $t->command }), "\n";

  155: sub new {
  156: 	my $class = shift;
  157: 	my $args = named(\@_, {
  158: 		map({ $_ => { type => 'string', optional => 1 } }
  159: 			qw(server repo client role server_command host_key_checking known_hosts identity port command)),
  160: 		extra_args => { type => 'arrayref', optional => 1 },
  161: 	}, 'Transport::SSH->new');
  162: 
  163: 	for my $required (qw(server repo client)) {
  164: 		throw('transport.missing', { transport => 'SSH', what => $required })
  165: 			unless defined $args->{$required} && length $args->{$required};
  166: 	}
  167: 	_check('server', $args->{server}, valid_server($args->{server}));
  168: 
  169: 	for my $check ([ repo => \&validate_repo_name ], [ client => \&validate_client_id ]) {
  170: 		my ($field, $validate) = @{$check};
  171: 		my ($ok, $reason) = $validate->($args->{$field});
  172: 		throw('transport.bad_id', { reason => $reason }) unless $ok;
  173: 	}
  174: 
  175: 	my $role = $args->{role} || $ROLE_CLIENT;
  176: 	_check('role', $role, any { $_ eq $role } @ROLES);
  177: 
  178: 	my $program = _given($args->{server_command}) // $SERVER_PROGRAM;
  179: 	_check('server_command', $program, valid_program($program));
  180: 
  181: 	my $checking = _given($args->{host_key_checking}) // $HOST_KEY_CHECKING[0];
  182: 	_check('host_key_checking', $checking, any { $_ eq $checking } @HOST_KEY_CHECKING);
  183: 
  184: 	my @command = (_given($args->{command}) // 'ssh', @SSH_OPTIONS, '-o', "StrictHostKeyChecking=$checking");
  185: 
  186: 	if (defined(my $known = _given($args->{known_hosts}))) {
  187: 		_check('known_hosts', $known, $known !~ $UNSAFE_ARG);
  188: 		push @command, '-o', "UserKnownHostsFile=$known";
  189: 	}
  190: 	if (defined(my $identity = _given($args->{identity}))) {
  191: 		_check('identity', $identity, $identity !~ $UNSAFE_ARG);
  192: 		push @command, '-i', $identity;
  193: 	}
  194: 	if (defined(my $port = _given($args->{port}))) {
  195: 		_check('port', $port, $port =~ /\A\d{1,5}\z/ && $port > 0 && $port <= $MAX_PORT);
  196: 		push @command, '-p', $port;
  197: 	}
  198: 	for my $extra (@{ $args->{extra_args} || [] }) {
  199: 		_check('extra_args', $extra // '', defined $extra && !ref $extra && $extra !~ $UNSAFE_ARG);
  200: 		push @command, $extra;
  201: 	}
  202: 
  203: 	# The remote shell splits this back into arguments; every value is
  204: 	# restricted (above) to characters that are inert in a POSIX shell.
  205: 	push @command, $args->{server}, $program,
  206: 		'--repo', $args->{repo}, '--client', $args->{client}, '--role', $role, '--proto', $PROTO;
  207: 
  208: 	if ($^O eq 'MSWin32' && $command[0] =~ /\.(?:cmd|bat)\z/i) {
  209: 		unshift @command, _given($ENV{COMSPEC}) // 'cmd.exe', '/d', '/c';
  210: 	}
  211: 
  212: 	return bless { command => \@command, status => undef, stderr => '', %{ _spawn(\@command) } }, $class;
  213: }

  331: sub close {	## no critic (Subroutines::ProhibitBuiltinHomonyms, Subroutines::ProhibitAmbiguousNames) -- transport interface
  332: 	my ($self) = @_;
  333: 
  334: 	return 1 if $self->{closed}++;
  335: 
  336: 	for my $handle (grep { defined } @{$self}{qw(writer reader)}) {
  337: 		CORE::close($handle);
  338: 	}
  339: 
  340: 	if ($self->{pid}) {
  341: 		waitpid($self->{pid}, 0);
  342: 		my $status = $?;
  343: 		@{$self}{qw(status signal pid)} = ($status >> 8, $status & 127, 0);
  344: 	}
  345: 	$self->{stderr} = _slurp($self->{error_path});
  346: 
  347: 	return ($self->{status} || 0) == 0 ? 1 : 0;
  348: }

  459: # _spawn
  460: # Purpose:     start ssh with stderr going to a temporary file.
  461: # Entry:       $command is the full argv.
  462: # Exit:        returns { pid, writer, reader, errors, error_path }.
  463: # Side effects: creates the error file (removed when the object goes) and
  464: #              starts the process.  IPC::Open3 rather than fork/exec: on
  465: #              Windows fork() is a pseudo-process whose pipe handles are
  466: #              not always duplicated, which leaves a full-duplex session
  467: #              waiting on a pipe end that never closes.
  468: sub _spawn :Private {
  469: 	my ($command) = @_;
  470: 
  471: 	my $errors = File::Temp->new(UNLINK => 1);
  472: 	my $error_path = $errors->filename;
  473: 	open my $err_fh, '>', $error_path;
  474: 
  475: 	my ($writer, $reader);
  476: 	my $pid = IPC::Open3::open3($writer, $reader, '>&' . fileno($err_fh), @{$command});
  477: 	CORE::close($err_fh);
  478: 
  479: 	return { pid => $pid, writer => $writer, reader => $reader, errors => $errors, error_path => $error_path };
  480: }
  481: 
  482: # _check
  483: # Purpose:     refuse an unsafe or out-of-range argument.
  484: # Entry:       $ok is the result of the check.
  485: # Exit:        returns 1.
  486: # Side effects: raises transport.bad_value naming the field.
  487: sub _check :Private {
  488: 	my ($field, $value, $ok) = @_;
  489: 
  490: 	throw('transport.bad_value', { field => $field, value => $value }) unless $ok;
  491: 	return 1;
  492: }
  493: 
  494: # _given
  495: # Purpose:     treat an empty string like an absent value.
  496: # Entry:       any scalar.
  497: # Exit:        returns the value, or undef if undefined or empty.
  498: # Side effects: none.
  499: sub _given :Private {
  500: 	my ($value) = @_;
  501: 
  502: 	return (defined $value && length $value) ? $value : undef;
  503: }
  504: 
  505: # _slurp
  506: # Purpose:     the error file's contents.
  507: # Entry:       $path may be undef or gone.
  508: # Exit:        returns the text, or ''.
  509: # Side effects: reads the file; an unreadable one is treated as empty
  510: #              (there is nothing better to report).
  511: sub _slurp :Private {
  512: 	my ($path) = @_;
  513: 
  514: 	return '' unless defined $path && -e $path;
  515: 	my $text = eval { read_text($path) };
  516: 	return defined $text ? $text : '';
  517: }
