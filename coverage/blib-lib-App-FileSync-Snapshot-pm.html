565# Entry:       $dir is the snapshot directory.
566# Exit:        returns 1.
567# Side effects: removes .tmp-PID-N directories older than an hour; younger
568#              ones may belong to a concurrent run.
569sub _clean_stale_temps :Private {
570        my ($dir) = @_;
571
572        my $now = time;
573        for my $name (grep { $_ =~ $TEMP_DIR } @{ list_dir($dir) || [] }) {
574                my $path = File::Spec->catdir($dir, $name);
575                my @stat = Time::HiRes::lstat($path);
576                next unless @stat && -d _ && $now - $stat[9] >= $STALE_TEMP_SECONDS;
577                eval { remove_tree($path) };
578        }
579
580        return 1;

623# _copy_out
624# Purpose:     copy a snapshot entry to its destination, replacing it.
625# Entry:       $src exists in the snapshot.
626# Exit:        returns 1.
627# Side effects: replaces $dst (unless $dry_run); updates $counts.
628sub _copy_out :Private {
629        my ($src, $dst, $counts, $dry_run) = @_;
630
631        my @stat = Time::HiRes::lstat($src);
632        throw('snapshot.cannot_stat', { path => $src, error => os_error() }) unless @stat;
633
634        if (-d _) {
635                ensure_dir($dst) unless $dry_run;
636                _copy_out(File::Spec->catfile($src, $_), File::Spec->catfile($dst, $_), $counts, $dry_run)
637                        for @{ list_dir($src) || [] };
638
639                # Directory metadata last: filling it changed its mtime.
640                unless ($dry_run) {
641                        chmod($stat[2] & $MODE_MASK, $dst) if $HAS_MODES;
642                        utime($stat[9], $stat[9], $dst);
643                }
644                return 1;
645        }
646
647        throw('snapshot.not_regular', { path => $src }) unless -f _;
648        $counts->{files}++;
649        $counts->{bytes} += $stat[7];
650        return 1 if $dry_run;
651
652        # A rollback replaces whatever is in the way, even a directory.
653        if (-d $dst && !-l $dst) {
654                remove_tree($dst);
655                throw('snapshot.replace', { path => $dst }) if -e $dst;
656        } elsif (-e $dst || -l $dst) {
657                eval { unlink $dst; 1 } or throw('snapshot.replace', { path => $dst });
658        }
659
660        copy_preserving($src, $dst, \@stat);
661        return 1;
