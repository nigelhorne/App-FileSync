3# rsync-style block matching.  The receiver describes the file it already has as
4# one strong hash per fixed-size block; the sender scans the new file with a
5# rolling checksum and emits either a copy of a block the receiver already has
6# or the literal bytes that changed.
7#
8# The scanner is pure Perl, so it is deliberately bounded: only files of at
9# least $MIN_SIZE are worth a signature, and at most $SCAN_LIMIT bytes per file
10# are examined for matches.  Whatever lies beyond that is sent literally, which
11# is still correct, just not smaller.  A small edit in a large file is cheap
12# because the unchanged leading blocks are matched without scanning.

61# rsync's weak checksum: two 16-bit sums, cheap to roll by one byte.  Lexical:
62# it runs once per block.
63#
64# b is defined as the sum of (length - i) * byte[i].  Adding each byte to a,
65# then a to b, gives the same total: byte[i] is in a for the length - i steps
66# from i on.  That needs no multiply and no index, and unpack walks the bytes
67# in C.  Unpacking a slice at a time keeps a large block from becoming one
68# huge list, and masking after each slice keeps both sums small integers
69# however long the block (the old per-byte product passed 2**63, and lost
70# precision, for blocks over about 256 MiB).  Masking early is safe: both
71# sums are only ever used modulo 2**16.
72my $weak_parts = sub {

Args: C<$fh>, C<$length>, C<$what> (named in errors).

Returns: the bytes.

Side Effects: advances C<$fh>.

Usage:

        my $data = App::FileSync::Delta::read_exact($fh, $block, $path);

=head3 EXAMPLE

        my $header = App::FileSync::Delta::read_exact($spool, 8, 'delta');

=head4 INPUT

                { name => 'fh',     type => 'handle' },
                { name => 'length', type => 'integer', min => 0 },
                { name => 'what',   type => 'string' },

=head4 OUTPUT

        { type => 'string' }

        io.short   fewer bytes than required       the file was truncated

730# _rolling_scan
731# Purpose:     find the receiver's blocks anywhere in the window.
732# Entry:       $buffer is the window; $blocks the receiver's signature.
733# Exit:        returns 1.
734# Side effects: writes copy and literal records for the whole buffer.
735sub _rolling_scan :Private {
736        my ($state, $buffer, $blocks) = @_;
737
738        my $block = $state->{block};
739        my $limit = length($buffer) - $block;
740
741        # Shorter than one block: nothing can match.
742        if ($limit < 0) {
743                _add_literal($state, $buffer);
744                return 1;
745        }
746
747        my %by_weak;
748        push @{ $by_weak{ $blocks->[$_][0] } }, [ $_, $blocks->[$_][1] ] for 0 .. $#{$blocks};
749
750        # A one-bit-per-value table of the low halves (a) of the receiver's weak
751        # sums, as rsync keeps.  Most windows fail this vec() test, which skips
752        # turning the sum into a string for the hash lookup.
753        my $tags = '';
754        vec($tags, $_->[0] & $HALF, 1) = 1 for @{$blocks};
755
756        # This loop runs once per byte that matches nothing, so its body calls no
757        # sub (the :Private wrapper costs about 1.4us a call).  Unmatched bytes are
758        # not queued one at a time: $run_start marks where the current literal run
759        # began, and the run is queued in one piece when a match or the end of the
760        # buffer closes it.  $a and $b are always already masked.
761        my $last = $limit > $SCAN_LIMIT ? $SCAN_LIMIT : $limit;
762        my ($a, $b) = $weak_parts->(substr($buffer, 0, $block));
763        my ($position, $run_start) = (0, 0);
764
765        while ($position <= $last) {
766                if (vec($tags, $a, 1) and my $candidates = $by_weak{ ($b << $HALF_BITS) | $a }) {
767                        my $winner = _strong_match($candidates, $buffer, $position, $block);
768                        if (defined $winner) {
769                                _add_literal($state, substr($buffer, $run_start, $position - $run_start)) if $position > $run_start;
770                                _add_copy($state, $winner * $block, $block);
771                                $position += $block;
772                                $run_start = $position;
773                                ($a, $b) = $weak_parts->(substr($buffer, $position, $block)) if $position <= $last;
774                                next;
775                        }
776                }
777
778                # Roll the window forward one byte, unless this was the last full
779                # window in the buffer.  The byte entering the window is the one just
780                # past its end.
781                if ($position < $last) {
782                        my $out_byte = vec($buffer, $position, $BYTE_BITS);
783                        my $in_byte  = vec($buffer, $position + $block, $BYTE_BITS);
784                        $a = ($a - $out_byte + $in_byte) & $HALF;
785                        $b = ($b - $block * $out_byte + $a) & $HALF;
786                }
787                $position++;
788        }
789
790        # The open run and everything past the last full window.
791        _add_literal($state, substr($buffer, $run_start)) if $run_start < length($buffer);
792        return 1;

795# _strong_match
796# Purpose:     confirm a weak-checksum hit with the strong hash.
797# Entry:       $candidates are the [ index, sha256 ] pairs whose weak sum
798#              matched the window at $position (the caller looks them up, so
799#              a byte with no weak hit costs no call).
800# Exit:        returns the matching block index, or undef.
801# Side effects: none.
802sub _strong_match :Private {
803        my ($candidates, $buffer, $position, $block) = @_;
804
805        my $strong = Digest::SHA::sha256(substr($buffer, $position, $block));
806        my ($hit) = grep { $_->[1] eq $strong } @{$candidates};
807
808        return $hit ? $hit->[0] : undef;

827# _add_literal
828# Purpose:     queue literal bytes, writing a record for each $LITERAL bytes
829#              that accumulate.
830# Entry:       $bytes may be empty, or a run of many megabytes.
831# Exit:        returns 1; fewer than $LITERAL bytes stay queued.
832# Side effects: may write to the output.  A long run is cut into $LITERAL
833#              records by offset, so the run is copied once, not once per
834#              record.
835sub _add_literal :Private {
836        my ($state, $bytes) = @_;
837
838        my $pending = $state->{literal} . $bytes;
839        my $offset = 0;
840        while (length($pending) - $offset >= $LITERAL) {
841                $state->{literal} = substr($pending, $offset, $LITERAL);
842                _flush($state);
843                $offset += $LITERAL;
844        }
845        $state->{literal} = substr($pending, $offset);
846
847        return 1;
