3# rsync-style block matching.  The receiver describes the file it already has as
4# one strong hash per fixed-size block; the sender scans the new file with a
5# rolling checksum and emits either a copy of a block the receiver already has
6# or the literal bytes that changed.
7#
8# The scanner is pure Perl, so it is deliberately bounded: only files of at
9# least $MIN_SIZE are worth a signature, and at most $SCAN_LIMIT bytes per file
10# are examined for matches.  Whatever lies beyond that is sent literally, which
11# is still correct, just not smaller.  A small edit in a large file is cheap
12# because the unchanged leading blocks are matched without scanning.

706# _rolling_scan
707# Purpose:     find the receiver's blocks anywhere in the window.
708# Entry:       $buffer is the window; $blocks the receiver's signature.
709# Exit:        returns 1.
710# Side effects: writes copy and literal records for the whole buffer.
711sub _rolling_scan :Private {
712        my ($state, $buffer, $blocks) = @_;
713
714        my $block = $state->{block};
715        my $limit = length($buffer) - $block;
716
717        # Shorter than one block: nothing can match.
718        if ($limit < 0) {
719                _add_literal($state, $buffer);
720                return 1;
721        }
722
723        my %by_weak;
724        push @{ $by_weak{ $blocks->[$_][0] } }, [ $_, $blocks->[$_][1] ] for 0 .. $#{$blocks};
725
726        my $last = $limit > $SCAN_LIMIT ? $SCAN_LIMIT : $limit;
727        my ($a, $b) = $weak_parts->(substr($buffer, 0, $block));
728        my $position = 0;

735                        $position += $block;
736                        ($a, $b) = $weak_parts->(substr($buffer, $position, $block)) if $position <= $last;
737                        next;
738                }
739
740                _add_literal($state, substr($buffer, $position, 1));
741
742                # Roll the window forward one byte, unless this was the last full
743                # window in the buffer.  The byte entering the window is the one just
744                # past its end.

747                        my $in_byte  = vec($buffer, $position + $block, 8);

749                        $b = ($b - $block * $out_byte + $a) & $HALF;
750                }
751                $position++;
752        }
